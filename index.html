function saveForm() {
const data = {};
document.querySelectorAll('input, select, textarea').forEach(el => {
if (el.id) data[el.id] = el.type === 'checkbox' ? el.checked : el.value;
});
const drawings = {};
['cTop', 'cSide', 'sigExec', 'sigClient'].forEach(id => {
if (canvases[id]) drawings[id] = canvases[id].canvas.toDataURL();
});
try {
localStorage.setItem('mmtarasy_pomiar', JSON.stringify({ fields: data, drawings }));
alert('Zapisano kopię roboczą (pola + szkice + podpisy)!');
} catch (e) {
alert('Zapisano pola formularza.');
}
}
function loadForm() {
try {
const raw = localStorage.getItem('mmtarasy_pomiar');
if (!raw) return;
const parsed = JSON.parse(raw);
const fields = parsed.fields || parsed;
Object.keys(fields).forEach(id => {
const el = document.getElementById(id);
if (!el) return;
if (el.type === 'checkbox') el.checked = fields[id];
else el.value = fields[id];
});
if (parsed.drawings) {
Object.keys(parsed.drawings).forEach(id => {
if (!canvases[id]) return;
const img = new Image();
img.onload = () => canvases[id].ctx.drawImage(img, 0, 0);
img.src = parsed.drawings[id];
});
}
} catch (e) {}
}
function resetAll() {
if (!confirm('Wyczyścić wszystkie dane i zacząć nowy pomiar?')) return;
try { localStorage.removeItem('mmtarasy_pomiar'); } catch (e) {}
document.querySelectorAll('input, select, textarea').forEach(el => {
if (el.type === 'checkbox') el.checked = false;
else if (el.type !== 'file') el.value = '';
});
document.getElementById('photoGrid').innerHTML = '';
['cTop', 'cSide', 'sigExec', 'sigClient'].forEach(clearCanvas);
document.getElementById('m_date').valueAsDate = new Date();
showPage(1);
}
window.addEventListener('DOMContentLoaded', () => {
document.getElementById('m_date').valueAsDate = new Date();
initCanvas('cTop', 2.5);
initCanvas('cSide', 2.5);
initCanvas('sigExec', 2.0);
initCanvas('sigClient', 2.0);
loadForm();
});
